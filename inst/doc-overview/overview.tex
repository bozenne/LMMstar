\section{Likelihood and derivatives in a linear mixed model}
\label{SM:likelihood}
In \texttt{lmm}:
\begin{itemize}
\item the argument \texttt{formula} specifies \(\mu(\Vparam,\VX)\) the modeled
mean of the outcome \(\VY\) conditional a vector of \(p\) covariates
\(\VX\).
\item the argument \texttt{repetition} (typically equal to \texttt{\textasciitilde{}time|id}) enable to
interpret \(\VY\) as a vector of \(m\) outcomes (indexed by \texttt{time})
for which we observe independent replicates (across \texttt{id}).
\item the argument \texttt{structure} define a model for the residual
variance-covariance \(\Omega(\Vparam,\VX)\).
\item the argument \texttt{weights} (typically equal to \texttt{r\textasciitilde{}w}) has on its
right-hand side a variable defining a vector of weights
\(\omega=(w_1,\ldots,w_n)\) - which should be constant within
cluster - and on its left hand side a variable defining the number
of replicates \(\mathbf{r}=(r_{1},\ldots,r_{T})\) which were
averaged to obtain the outcome \(\VY\). When left un-specified, the
weights and number of replicates are considered equal to 1.
\end{itemize}
\subsection{Log-likelihood}
\label{SM:likelihood:log}
With \(n\) independent replicates \(\left(\VY_1,\ldots,\VY_n\right)\) and
\(\left(\VX_1,\ldots,\VX_n\right)\), the restricted Gaussian log-likelihood is:
\begin{align}
\Likelihood(\Vparam|\VY,\VX) =& \textcolor{\darkred}{ \frac{p}{2} \log(2\pi)-\frac{1}{2} \log\left(\left|\sum_{i=1}^n w_i \VX_i \Omega_i^{-1}(\Vparam) \trans{\VX}_i\right|\right)} \notag \\
& + \sum_{i=1}^{n} w_i \left(\textcolor{\darkblue}{-\frac{m}{2} \log(2\pi) - \frac{1}{2} \log\left|\Omega_i(\Vparam)\right| - \frac{1}{2} \Vvarepsilon_i(\Vparam) \Omega_i(\Vparam)^{-1} \trans{\Vvarepsilon_i(\Vparam)}} \right)  \label{eq:log-likelihood}
\end{align}

where \(\Vvarepsilon_i(\Vparam) = \VY_i-\mu(\Vparam,\VX_i)\) denotes
 the residuals. This is what the \texttt{logLik} method is computing for the
 REML criteria. The red term is specific to the REML criteria and
 prevents from computing individual contributions to the
 likelihood\footnote{The REML is the likelihood of the observations divided
 by the prior on the estimated mean parameters \(\VparamHat_{\mu} \sim
 \Gaus(\mu,\left(\VX \Omega^{-1}(\Vparam)
 \trans{\VX}\right)^{-1})\). This corresponds to
 \(\frac{1}{\sqrt{2\pi}^p \left|\left(\sum_{i=1}^n \VX_i
 \Omega_i^{-1}(\Vparam) \trans{\VX}_i\right)^{-1}\right|}
 \exp\left(-(\VparamHat_{\mu}-\mu)\left(2\sum_{i=1}^n \VX_i
 \Omega_i^{-1}(\Vparam)
 \trans{\VX}_i\right)^{-1}\trans{(\VparamHat_{\mu}-\mu)}\right)\)
 Since \(\mu\) will be estimated to be \(\Vparam_{\mu}\), the
 exponential term equals 1 and thus does not contribute to the
 log-likelihood. One divided by the other term gives \(\sqrt{2\pi}^p
 \left(\left|\sum_{i=1}^n \VX_i \Omega_i^{-1}(\Vparam)
 \trans{\VX}_i\right|\right)^{-1}\). The log of this term equals the
 red term.}. The blue term is what \texttt{logLik} outputs for the ML criteria
 when setting the argument \texttt{indiv} to \texttt{TRUE}.

\bigskip

\uline{Note:} the implementation uses that within a subset of subject sharing the same covariance pattern:
\begin{align*}
\sum_{i} w_i \VX_i \Omega(\Vparam)^{-1} \trans{\VX_i}
&= \trans{\left(\sum_{i} w_i \VX_i \otimes \VX_i\right)} \text{vec}\left(\Omega(\Vparam)^{-1}\right) \\
\sum_{i} w_i \Vvarepsilon_i(\Vparam) \Omega(\Vparam)^{-1} \trans{\Vvarepsilon_i(\Vparam)}
&= \trans{\left(\sum_{i} w_i \Vvarepsilon_i(\Vparam) \otimes \Vvarepsilon_i(\Vparam)\right)} \text{vec}\left(\Omega(\Vparam)^{-1}\right) 
\end{align*}
where \(\otimes\) denotes the kronecker product \(vec\) the
vectorization of a matrix. \(\sum_{i} w_i \VX_i
\otimes \VX_i = \sum_{i} \VX^*_i
\otimes \VX^*_i\) can then be pre-computed with \(\VX^*_i=\sqrt{w_i}\VX_i\) and
does not need to be updated during the optimization process.

\subsection{Score}
\label{sec:org020f7ee}

 Using that \(\partial \log(\det(X))=tr(X^{-1}\partial(X))\), the
score is obtained by derivating once the log-likelihood, i.e., for
\(\theta \in \Vparam\):
\begin{align*}
   &\Score(\theta) = \dpartial[\Likelihood(\Vparam|\VY,\VX)][\theta]
= \textcolor{\darkred}{ \frac{1}{2} tr \left( \left(\sum_{j=1}^n w_j \VX_j \Omega_j^{-1}(\Vparam) \trans{\VX}_j\right)^{-1} \left(\sum_{i=1}^n w_i \VX_i \Omega_i^{-1}(\Vparam) \dpartial[\Omega_i(\Vparam)][\theta] \Omega_i(\Vparam)^{-1} \trans{\VX}_i\right)  \right) } \\
&+ \sum_{i=1}^n w_i \left( \textcolor{\darkblue}{ -\frac{1}{2} tr\left(\Omega_i(\Vparam)^{-1} \dpartial[\Omega_i(\Vparam)][\theta]\right) + \dpartial[\mu(\Vparam,\VX_i)][\theta] \Omega_i(\Vparam)^{-1} \trans{\Vvarepsilon_i(\Vparam)} 
+ \frac{1}{2} \Vvarepsilon_i(\Vparam) \Omega_i(\Vparam)^{-1} \dpartial[\Omega_i(\Vparam)][\theta] \Omega_i(\Vparam)^{-1} \trans{\Vvarepsilon_i(\Vparam)} } \right).
\end{align*}

This is what the \texttt{score} method is computing for the REML criteria
(blue and red) and the ML (blue only) criteria. When the individual
contribution is requested (\texttt{indiv=TRUE}), \texttt{score} output:
\begin{align*}
   \Score_i(\theta) &
= \textcolor{\darkred}{ \frac{1}{2} tr \left( \left(\sum_{j=1}^n w_j \VX_j \Omega_j^{-1}(\Vparam) \trans{\VX}_j\right)^{-1} \left(w_i \VX_i \Omega_i^{-1}(\Vparam) \dpartial[\Omega_i(\Vparam)][\theta] \Omega_i(\Vparam)^{-1} \trans{\VX}_i\right)  \right) } \\
&+ w_i \left( \textcolor{\darkblue}{ -\frac{1}{2} tr\left(\Omega_i(\Vparam)^{-1} \dpartial[\Omega_i(\Vparam)][\theta]\right) + \dpartial[\mu(\Vparam,\VX_i)][\theta] \Omega_i(\Vparam)^{-1} \trans{\Vvarepsilon_i(\Vparam)}
+ \frac{1}{2} \Vvarepsilon_i(\Vparam) \Omega_i(\Vparam)^{-1} \dpartial[\Omega_i(\Vparam)][\theta] \Omega_i(\Vparam)^{-1} \trans{\Vvarepsilon_i(\Vparam)} } \right).
\end{align*}
\subsection{Hessian}
\label{SM:likelihood:hessian}
Derivating a second time the log-likelihood gives the hessian,
\(\Hessian(\Vparam)\), with element:
\begin{align*}
& \Hessian(\theta,\theta^{\prime}) = \ddpartial[\Likelihood(\Vparam|\VY,\VX)][\theta][\theta^{\prime}] = \dpartial[\Score(\theta)][\theta^{\prime}] \\
=& \textcolor{\darkred}{\frac{1}{2} tr \left( \left(\sum_{j=1}^n w_j \VX_j \Omega_j^{-1}(\Vparam) \trans{\VX}_j\right)^{-1} \left\{ \sum_{i=1}^n w_i \VX_i \Omega_i^{-1}(\Vparam) \left(\ddpartial[\Omega_i(\Vparam)][\theta][\theta^{\prime}] - 2 \dpartial[\Omega_i(\Vparam)][\theta] \Omega_i^{-1}(\Vparam) \dpartial[\Omega_i(\Vparam)][\theta^{\prime}]\right)\Omega_i(\Vparam)^{-1} \trans{\VX}_i \right.  \right.}  \\
& \textcolor{\darkred}{ \left. \left. + \left(\sum_{i=1}^n w_i \VX_i \Omega_i^{-1}(\Vparam) \dpartial[\Omega_i(\Vparam)][\theta] \Omega_i(\Vparam)^{-1} \trans{\VX}_i\right) \left(\sum_{j=1}^n w_j \VX_i\Omega_j^{-1}(\Vparam) \trans{\VX}_j \right)^{-1} \left(\sum_{i=1}^n w_i \VX_i \Omega_i^{-1}(\Vparam) \dpartial[\Omega_i(\Vparam)][\theta^{\prime}] \Omega_i(\Vparam)^{-1} \trans{\VX}_i\right) \right\} \right) } \\
& +\sum_{i=1}^n w_i \left( \textcolor{\darkblue}{ \frac{1}{2} tr\left(\Omega_i(\Vparam)^{-1} \dpartial[\Omega_i(\Vparam)][\theta^{\prime}] \Omega_i(\Vparam)^{-1} \dpartial[\Omega_i(\Vparam)][\theta] - \Omega_i(\Vparam)^{-1} \ddpartial[\Omega_i(\Vparam)][\theta][\theta^{\prime}] \right) } \right.\\
& \qquad \textcolor{\darkblue}{ -  \dpartial[\mu(\Vparam,\VX_i)][\theta] \Omega_i(\Vparam)^{-1} \dpartial[\Omega_i(\Vparam)][\theta^{\prime}] \Omega_i(\Vparam)^{-1} \trans{\Vvarepsilon_i(\Vparam)} - \dpartial[\mu(\Vparam,\VX_i)][\theta] \Omega_i(\Vparam)^{-1} \trans{\dpartial[\mu(\Vparam,\VX_i)][\theta^{\prime}]} } \\
& \qquad \left. \textcolor{\darkblue}{ + \frac{1}{2} \Vvarepsilon_i(\Vparam) \Omega_i(\Vparam)^{-1} \left(\ddpartial[\Omega_i(\Vparam)][\theta][\theta^{\prime}] - \dpartial[\Omega_i(\Vparam)][\theta^{\prime}] \Omega_i(\Vparam)^{-1} \dpartial[\Omega_i(\Vparam)][\theta] - \dpartial[\Omega_i(\Vparam)][\theta] \Omega_i(\Vparam)^{-1} \dpartial[\Omega_i(\Vparam)][\theta^{\prime}] \right) \Omega_i(\Vparam)^{-1} \trans{\Vvarepsilon_i(\Vparam)} } \right).
\end{align*}
where \((\theta,\theta^{\prime})\) are either two mean parameters, or
 two variance parameters, or \(\theta\) is a mean parameter and
 \(\theta^{\prime}\) as variance parameter. The case \(\theta\)
 variance parameter and \(\theta\) mean parameter can be deduced by
 symmetry; doing so enable some simplifications since
 \(\dpartial[\mu(\Vparam,\VX_i)][\theta^{\prime}]\dpartial[\Omega(\Vparam,\VX_i)][\theta]=0\).

\bigskip

The \texttt{information} method will (by default) return the (observed)
 information which is the opposite of the hessian. So the previous
 formula shows \textbf{-1} times what the \texttt{information} method is computing
 for the REML criteria (blue and red) and the ML (blue only) criteria,
 with the default option
 \texttt{type.information="observed"}. When the individual
 contribution is requested (\texttt{indiv=TRUE}), \texttt{information} output:
\begin{align*}
& \Hessian_i(\theta,\theta^{\prime}) = \textcolor{\darkred}{\frac{1}{2} tr \left( \left(\sum_{j=1}^n w_j \VX_j \Omega_j^{-1}(\Vparam) \trans{\VX}_j\right)^{-1} \left\{w_i \VX_i \Omega_i^{-1}(\Vparam) \left(\ddpartial[\Omega_i(\Vparam)][\theta][\theta^{\prime}] - 2 \dpartial[\Omega_i(\Vparam)][\theta] \Omega_i^{-1}(\Vparam) \dpartial[\Omega_i(\Vparam)][\theta^{\prime}]\right)\Omega_i(\Vparam)^{-1} \trans{\VX}_i \right.  \right.}  \\
& \textcolor{\darkred}{ \left. \left. + 0.5\left(w_i \VX_i \Omega_i^{-1}(\Vparam) \dpartial[\Omega_i(\Vparam)][\theta] \Omega_i(\Vparam)^{-1} \trans{\VX}_i\right) \left(\sum_{j=1}^n w_j \VX_i\Omega_j^{-1}(\Vparam) \trans{\VX}_j \right)^{-1} \left(\sum_{j=1}^n w_j \VX_j \Omega_j^{-1}(\Vparam) \dpartial[\Omega_j(\Vparam)][\theta^{\prime}] \Omega_j(\Vparam)^{-1} \trans{\VX}_j\right) \right\} \right) } \\
& \textcolor{\darkred}{ \left. \left. + 0.5\left(\sum_{j=1}^n w_j \VX_j \Omega_j^{-1}(\Vparam) \dpartial[\Omega_j(\Vparam)][\theta] \Omega_j(\Vparam)^{-1} \trans{\VX}_j\right) \left(\sum_{j=1}^n w_j \VX_i\Omega_j^{-1}(\Vparam) \trans{\VX}_j \right)^{-1} \left(w_i \VX_i \Omega_i^{-1}(\Vparam) \dpartial[\Omega_i(\Vparam)][\theta^{\prime}] \Omega_i(\Vparam)^{-1} \trans{\VX}_i\right) \right\} \right) } \\
& + w_i \left( \textcolor{\darkblue}{ \frac{1}{2} tr\left(\Omega_i(\Vparam)^{-1} \dpartial[\Omega_i(\Vparam)][\theta^{\prime}] \Omega_i(\Vparam)^{-1} \dpartial[\Omega_i(\Vparam)][\theta] - \Omega_i(\Vparam)^{-1} \ddpartial[\Omega_i(\Vparam)][\theta][\theta^{\prime}] \right) } \right.\\
& \qquad \textcolor{\darkblue}{ -  \dpartial[\mu(\Vparam,\VX_i)][\theta] \Omega_i(\Vparam)^{-1} \dpartial[\Omega_i(\Vparam)][\theta^{\prime}] \Omega_i(\Vparam)^{-1} \trans{\Vvarepsilon_i(\Vparam)} - \dpartial[\mu(\Vparam,\VX_i)][\theta] \Omega_i(\Vparam)^{-1} \trans{\dpartial[\mu(\Vparam,\VX_i)][\theta^{\prime}]} } \\
& \qquad \left. \textcolor{\darkblue}{ + \frac{1}{2} \Vvarepsilon_i(\Vparam) \Omega_i(\Vparam)^{-1} \left(\ddpartial[\Omega_i(\Vparam)][\theta][\theta^{\prime}] - \dpartial[\Omega_i(\Vparam)][\theta^{\prime}] \Omega_i(\Vparam)^{-1} \dpartial[\Omega_i(\Vparam)][\theta] - \dpartial[\Omega_i(\Vparam)][\theta] \Omega_i(\Vparam)^{-1} \dpartial[\Omega_i(\Vparam)][\theta^{\prime}] \right) \Omega_i(\Vparam)^{-1} \trans{\Vvarepsilon_i(\Vparam)} } \right).
\end{align*}
\subsection{Expected Hessian}
\label{SM:likelihood:Ehessian}
A possible simplification is to use the expected Hessian at the Maximum (Restricted) Likelihood. Indeed for
any deterministic matrix \(A\):
\begin{itemize}
\item \(\Esp[A \trans{(\VY_i-\mu(\Vparam,\VX_i))}|\VX_i] = 0\)
\item \(\Esp[(\VY_i-\mu(\Vparam,\VX_i)) A \trans{(\VY_i-\mu(\Vparam,\VX_i))}||\VX_i] = tr(A \Var(\VY_i-\mu(\Vparam,\VX_i)))\)
\end{itemize}
when \(\Esp[\VY_i-\mu(\Vparam,\VX_i)]=0\). This leads to:
\begin{align}
 & \Esp[\Hessian(\theta,\theta^{\prime})|\VX] \notag\\ 
 &= \textcolor{\darkred}{ \frac{1}{2} tr \left( \left(\sum_{i=1}^n w_i \VX_i \Omega_i^{-1}(\Vparam) \trans{\VX}_i\right)^{-1}  \left\{ \sum_{i=1}^n w_i \VX_i \Omega_i^{-1}(\Vparam) \left( \ddpartial[\Omega_i(\Vparam)][\theta][\theta^{\prime}] - 2 \dpartial[\Omega_i(\Vparam)][\theta]  \Omega_i^{-1}(\Vparam) \dpartial[\Omega_i(\Vparam)][\theta^{\prime}]\right) \Omega_i(\Vparam)^{-1} \trans{\VX}_i \right.  \right.} \notag \\
 & \textcolor{\darkred}{ \left. \left. +  \left(\sum_{i=1}^n w_i \VX_i \Omega_i^{-1}(\Vparam) \dpartial[\Omega_i(\Vparam)][\theta] \Omega_i(\Vparam)^{-1} \trans{\VX}_i\right) \left(\sum_{i=1}^n w_i \VX_i \Omega_i^{-1}(\Vparam) \trans{\VX}_i \right)^{-1} \left(\sum_{i=1}^n w_i \VX_i \Omega_i^{-1}(\Vparam) \dpartial[\Omega_i(\Vparam)][\theta^{\prime}] \Omega_i(\Vparam)^{-1} \trans{\VX}_i\right) \right\} \right) } \notag\\
 & + \sum_{i=1}^n w_i \left( \textcolor{\darkblue}{
- \frac{1}{2} tr\left(\Omega_i(\Vparam)^{-1} \dpartial[\Omega_i(\Vparam)][\theta^{\prime}] \Omega_i(\Vparam)^{-1} \dpartial[\Omega_i(\Vparam)][\theta]\right)
 - \dpartial[\mu(\Vparam,\VX_i)][\theta] \Omega_i(\Vparam)^{-1} \trans{\dpartial[\mu(\Vparam,\VX_i)][\theta^{\prime}]}
 } \right) \label{eq:expectedInfo}
\end{align}

which is what \texttt{information} output when the argument
\texttt{type.information} is set to \texttt{"expected"} (up to a factor \textbf{-1}). A
similar simplification is used when \texttt{indiv} is set to \texttt{TRUE}.

\subsection{Degrees of freedom}
\label{sec:orgbda6280}

Degrees of freedom are computed using a Satterthwaite approximation,
i.e. for an estimate coefficient \(\widehat{\beta}\in\widehat{\Vparam}\) with standard
error \(\sigma_{\widehat{\beta}}\), the degree of freedom is:
\begin{align*}
df\left(\sigma_{\widehat{\beta}}\right) = \frac{2 \sigma^4_{\widehat{\beta}}}{\Var[\widehat{\sigma}_{\widehat{\beta}}]}
\end{align*}
Using a first order Taylor expansion we can approximate the variance term as:
\begin{align*}
\Var[\widehat{\sigma}_{\widehat{\beta}}] & \approx \dpartial[\widehat{\sigma}_{\widehat{\beta}}][\Vparam] \Sigma_{\Vparam}  \trans{\dpartial[\widehat{\sigma}_{\widehat{\beta}}][\Vparam]} \\
& \approx c_{\beta} \left(\widehat{\Information}_{\widehat{\Vparam}}\right)^{-1} \dpartial[\widehat{\Information}_{\widehat{\Vparam}}][\Vparam] \left(\widehat{\Information}_{\widehat{\Vparam}}\right)^{-1} \trans{c_{\beta}} \Sigma_{\Vparam} \trans{c_{\beta}} \left(\widehat{\Information}_{\widehat{\Vparam}}\right)^{-1} \trans{\dpartial[\widehat{\Information}_{\widehat{\Vparam}}][\Vparam]} \left(\widehat{\Information}_{\widehat{\Vparam}}\right)^{-1} c_{\beta}
\end{align*}

where \(\Sigma_{\Vparam}\) is the variance-covariance matrix of all
  model coefficients, \(\Information_{\Vparam}\) the information
  matrix for all model coefficients, \(c_{\beta}\) a matrix used to
  select the element relative to \(\beta\) in the first derivative of
  the information matrix, and \(\dpartial[.][\Vparam]\) denotes the
  vector of derivatives with respect to all model coefficients.

\bigskip

For the derivative of the information (i.e. \textbf{-1} times the derivative
 of the hessian) we consider
 \((\theta,\theta^{\prime},,\theta^{\prime\prime})\) are either three
 mean parameters, or three variance parameters, or \(\theta\) is a
 mean parameter and \(\theta^{\prime}\) and \(\theta^{\prime\prime}\)
 are variance parameters, or \(\theta\) and \(\theta^{\prime}\) are
 mean parameters and \(\theta^{\prime\prime}\) is a variance
 parameter. The other cases follow by symmetry; doing so enable some
 simplifications since
 \(\dpartial[\mu(\Vparam,\VX_i)][\theta^{\prime}]\dpartial[\Omega(\Vparam,\VX_i)][\theta^{\prime}]\dpartial[\mu(\Vparam,\VX_i)][\theta^{\prime\prime}]=0\). We
 first focus on the derivative of the ML term:

\begin{align*}
& \dpartial[\textcolor{\darkblue}{\Hessian(\theta,\theta^{\prime})}][\theta^{\prime\prime}]
 = \sum_{i=1}^n w_i \left( \textcolor{\darkblue}{
 \frac{1}{2} tr\left(\Omega_i(\Vparam)^{-1} \dpartial[\Omega_i(\Vparam)][\theta^{\prime\prime}] \Omega_i(\Vparam)^{-1} \ddpartial[\Omega_i(\Vparam)][\theta][\theta^{\prime}] - \Omega_i(\Vparam)^{-1} \dddpartial[\Omega_i(\Vparam)][\theta][\theta^{\prime}][\theta^{\prime\prime}] - \Upsilon_{\theta\theta^{\prime}\theta^{\prime\prime}} - \Upsilon_{\theta^{\prime}\theta\theta^{\prime\prime}} \right)
 } \right. \\
& \qquad \textcolor{\darkblue}{+  \dpartial[\mu(\Vparam,\VX_i)][\theta] \Omega_i(\Vparam)^{-1}
\left( \dpartial[\Omega_i(\Vparam)][\theta^{\prime\prime}] \Omega_i(\Vparam)^{-1} \dpartial[\Omega_i(\Vparam)][\theta^{\prime}] - \ddpartial[\Omega_i(\Vparam)][\theta^{\prime}][\theta^{\prime\prime}] + \dpartial[\Omega_i(\Vparam)][\theta^{\prime}] \Omega_i(\Vparam)^{-1} \dpartial[\Omega_i(\Vparam)][\theta^{\prime\prime}] \right)
\Omega_i(\Vparam)^{-1} \trans{\Vvarepsilon_i(\Vparam)} }\\
& \qquad \textcolor{\darkblue}{
+   \dpartial[\mu(\Vparam,\VX_i)][\theta] \Omega_i(\Vparam)^{-1} \dpartial[\Omega_i(\Vparam)][\theta^{\prime\prime}] \Omega_i(\Vparam)^{-1} \trans{\dpartial[\mu(\Vparam,\VX_i)][\theta^{\prime}]}
} \\
& \qquad \left. \textcolor{\darkblue}{ + \frac{1}{2} \Vvarepsilon_i(\Vparam) \left(
\Omega_i(\Vparam)^{-1} \dddpartial[\Omega_i(\Vparam)][\theta][\theta^{\prime}][\theta^{\prime\prime}]
+ \Upsilon_{\theta\theta^{\prime}\theta^{\prime\prime}}
+ \Upsilon_{\theta\theta^{\prime\prime}\theta^{\prime}}
+ \Upsilon_{\theta^{\prime}\theta\theta^{\prime\prime}}
+ \Upsilon_{\theta^{\prime}\theta^{\prime\prime}\theta}
+ \Upsilon_{\theta^{\prime\prime}\theta\theta^{\prime}}
+ \Upsilon_{\theta^{\prime\prime}\theta^{\prime}\theta}
\right)\Omega_i(\Vparam)^{-1} \trans{\Vvarepsilon_i(\Vparam)} } \right) .
\end{align*}
where
\begin{align*}
\Upsilon_{\theta\theta^{\prime}\theta^{\prime\prime}} =& \Omega_i(\Vparam)^{-1} \dpartial[\Omega_i(\Vparam)][\theta]  \Omega_i(\Vparam)^{-1} \left(\dpartial[\Omega_i(\Vparam)][\theta^{\prime}] \Omega_i(\Vparam)^{-1} \dpartial[\Omega_i(\Vparam)][\theta^{\prime\prime}] - \ddpartial[\Omega_i(\Vparam)][\theta^{\prime}][\theta^{\prime\prime}]\right) \\
\Upsilon_{\theta^{\prime}\theta\theta^{\prime\prime}} =& \Omega_i(\Vparam)^{-1} \dpartial[\Omega_i(\Vparam)][\theta^{\prime}] \Omega_i(\Vparam)^{-1} \left(\dpartial[\Omega_i(\Vparam)][\theta] \Omega_i(\Vparam)^{-1} \dpartial[\Omega_i(\Vparam)][\theta^{\prime\prime}] - \ddpartial[\Omega_i(\Vparam)][\theta][\theta^{\prime\prime}]\right) \\
\Upsilon_{\theta^{\prime\prime}\theta\theta^{\prime}} =& \Omega_i(\Vparam)^{-1} \dpartial[\Omega_i(\Vparam)][\theta^{\prime\prime}] \Omega_i(\Vparam)^{-1} \left(\dpartial[\Omega_i(\Vparam)][\theta] \Omega_i(\Vparam)^{-1} \dpartial[\Omega_i(\Vparam)][\theta^{\prime}] - \ddpartial[\Omega_i(\Vparam)][\theta][\theta^{\prime}]\right) 
\end{align*}
In the special case of the expected information, the derivative of the ML term simplifies to:
\begin{align*}
\dpartial[\textcolor{\darkblue}{\Esp\left[\Hessian(\theta,\theta^{\prime})|\VX\right]}][\theta^{\prime\prime}]=
 \sum_{i=1}^n w_i & \left( \textcolor{\darkblue}{
\dpartial[\mu(\Vparam,\VX_i)][\theta] \Omega_i(\Vparam)^{-1} \dpartial[\Omega_i(\Vparam)][\theta^{\prime\prime}] \Omega_i(\Vparam)^{-1}   \trans{\dpartial[\mu(\Vparam,\VX_i)][\theta^{\prime}]}
+ \frac{1}{2} tr\left(\Upsilon_{\theta\theta^{\prime}\theta^{\prime\prime}} + \Upsilon_{\theta^{\prime}\theta\theta^{\prime\prime}} \right)
} \right)
\end{align*}
The REML term can be re-expressed as:
\begin{align*}
\textcolor{\darkred}{\Hessian(\theta,\theta^{\prime})} &= \textcolor{\darkred}{
\frac{1}{2} tr \left(A^{-1} \left( C_{\theta\theta^{\prime}} + B_{\theta}A^{-1} B_{\theta^{\prime}} \right)\right)
} \\
\textcolor{\darkred}{A} &= \textcolor{\darkred}{\sum_{i=1}^n w_i \VX_i \Omega_i^{-1}(\Vparam) \trans{\VX}_i }\\
\textcolor{\darkred}{B_{\theta}} &= \textcolor{\darkred}{\sum_{i=1}^n w_i \VX_i \Omega_i^{-1}(\Vparam) \dpartial[\Omega_i(\Vparam)][\theta] \Omega_i(\Vparam)^{-1} \trans{\VX}_i } \\
\textcolor{\darkred}{C_{\theta\theta^{\prime}}} &= \textcolor{\darkred}{\sum_{i=1}^n w_i \VX_i \Omega_i^{-1}(\Vparam)\left(\ddpartial[\Omega_i(\Vparam)][\theta][\theta^{\prime}] - 2 \dpartial[\Omega_i(\Vparam)][\theta]  \Omega_i^{-1}(\Vparam) \dpartial[\Omega_i(\Vparam)][\theta^{\prime}]\right) \Omega_i^{-1}(\Vparam) \trans{\VX}_i }
\end{align*}
Therefore the third derivative of the REML term \(\textcolor{\darkred}{\dpartial[\Hessian(\theta,\theta^{\prime})][\theta^{\prime\prime}]}\) equals: 
\begin{align*}
&\textcolor{\darkred}{
\frac{1}{2} tr \left(A^{-1} \left(
D_{\theta\theta^{\prime}\theta^{\prime\prime}} + B_{\theta^{\prime\prime}} A^{-1} C_{\theta\theta^{\prime}} + C_{\theta\theta^{\prime\prime}} A^{-1} B_{\theta^{\prime}} + B_{\theta}A^{-1} C_{\theta^{\prime}\theta^{\prime\prime}} + B_{\theta^{\prime\prime}}A^{-1}B_{\theta}A^{-1}B_{\theta^{\prime}} + B_{\theta}A^{-1}B_{\theta^{\prime\prime}}A^{-1}B_{\theta^{\prime}}
\right)\right) } \\
&\text{ with }\textcolor{\darkred}{D_{\theta\theta^{\prime}\theta^{\prime\prime}}} = \textcolor{\darkred}{\sum_{i=1}^n w_i \VX_i \left(
\Omega_i(\Vparam)^{-1} \dddpartial[\Omega_i(\Vparam)][\theta][\theta^{\prime}][\theta^{\prime\prime}]
+ \Upsilon_{\theta\theta^{\prime}\theta^{\prime\prime}}
+ \Upsilon_{\theta\theta^{\prime\prime}\theta^{\prime}}
+ \Upsilon_{\theta^{\prime}\theta\theta^{\prime\prime}}
+ \Upsilon_{\theta^{\prime}\theta^{\prime\prime}\theta}
+ \Upsilon_{\theta^{\prime\prime}\theta\theta^{\prime}}
+ \Upsilon_{\theta^{\prime\prime}\theta^{\prime}\theta}
\right)\Omega_i(\Vparam)^{-1}  \trans{\VX}_i }
\end{align*}
\subsection{Averaging over replicates}
\label{sec:org27a3135}

Aeraging over technical replicates \(\VY^*\) to obtain \(\VY\)
rescales the residual variance-covariance matrix. Denote by
\(\Omega_i(\Vparam)\) the 'single replicate' residual
variance-covariance matrix which is to be specified in argument
\texttt{structure} and by \(\overline{\Omega}_i(\Vparam)\) the
variance-covariance matrix of the average replicate. Assuming
independent replicates, i.e.  for two replicates \(r\) and
\(r^{\prime}\):
\(\Cov(\VY^*_{ir},\VY^*_{ir^{\prime}})=\Omega_i(\Vparam)
\Ind[r=r^{\prime}]\), we can relate \(\overline{\Omega}_i(\Vparam)\)
to \(\Omega_i(\Vparam)\):
\begin{align*}
\overline{\Omega}_{itt^{\prime}}(\Theta) = \Cov\left(Y_{it},Y_{it^{\prime}}\right)=\frac{1}{r_{it}r_{it^{\prime}}}\sum_{r=1}^{r_{it}}\sum_{r^{\prime}=1}^{r_{it^{\prime}}}\Cov\left(Y^*_{itr},Y^*_{it^{\prime}r^{\prime}}\right)
=\frac{\min(r_{it},r_{it^{\prime}})}{r_{it}r_{it^{\prime}}}\Omega_{itt^{\prime}}(\Theta)
=\frac{1}{\max(r_{it},r_{it^{\prime}})}\Omega_{itt^{\prime}}(\Theta)
\end{align*}

Denote by \(r_{i\pi(1)},\ldots,r_{i\pi(T)}\) the number of replicates
per outcome for subject \(i\) sorted in decreasing order. Introduce:
\begin{itemize}
\item \(\tau=\begin{bmatrix}1 & 0 & \ldots & 0 & 0 \\ 1 & 1 & \ldots & 0 &
  0 \\ \vdots & \vdots & \ddots & \vdots & \vdots \\ 1 & 1 & \ldots &
  1 & 0 \\ 1 & 1 & \ldots & 1 & 1 \\ \end{bmatrix}\) the 'unit' lower
triangular matrix
\item \(\delta(\mathbf{r}_i)=\begin{bmatrix}r_{i\pi(1)} & 0 & \ldots & 0 &
  0 \\ 0 & r_{i\pi(2)} - r_{i\pi(1)} & \ldots & 0 & 0 \\ \vdots &
  \vdots & \ddots & \vdots & \vdots \\ 0 & 0 & \ldots &
  r_{i\pi(T-2)} - r_{i\pi(T-1)} & 0 \\ 0 & 0 & \ldots & 0 &
  r_{i\pi(T-1)} - r_{i\pi(T)} \\ \end{bmatrix}\) the diagonal matrix
of the difference in number of replicates
\item \(P\) the permutation matrix with element \(P_{st}=\Ind[s=\pi(t)]\)
\end{itemize}
Then, denoting by \(\odot\) the element-wise product (also called Hadamard product):  
\begin{align*}
\overline{\Omega}_{i}(\Vparam) = \Omega_i(\Vparam) \odot (P \tau)  \delta(\mathbf{r}_i) \trans{(P \tau)}
\end{align*}
leading to following log-likelihood:
\begin{align*}
\Likelihood(\Vparam|\VY,\VX) =& \textcolor{\darkred}{ \frac{p}{2} \log(2\pi)-\frac{1}{2} \log\left(\left|\sum_{i=1}^n w_i \VX_i \overline{\Omega}_i^{-1}(\Vparam) \trans{\VX}_i\right|\right)} \notag \\
& + \sum_{i=1}^{n} w_i \left(\textcolor{\darkblue}{-\frac{m}{2} \log(2\pi) - \frac{1}{2} \log\left|\overline{\Omega}_i(\Vparam)\right| - \frac{1}{2} \Vvarepsilon_i(\Vparam) \overline{\Omega}_i(\Vparam)^{-1} \trans{\Vvarepsilon_i(\Vparam)}} \right) 
\end{align*}
The expression of the score, Hessian, degrees of freedom can be
modified similarly substituting \(\overline{\Omega}\) to \(\Omega\)
where for the derivatives one should take the element-wise product with
the replication factor, e.g.:
\begin{align*}
\dpartial[\overline{\Omega}_{i}(\Vparam)][\theta] = \dpartial[\Omega_{i}(\Vparam)][\theta] \odot (P \tau)  \delta(\mathbf{r}_i) \trans{(P \tau)}
\end{align*}


\textbf{Special case} (\(r_{it}=r_{i}\)): number of replicates is constant
within subjects across outcomes (but may vary across subjects). Then
\(\overline{\Omega}_{i}(\Vparam) =
\frac{\Omega_i(\Vparam)}{r_{i}}\). With \(w_i^*=w_i r_i\), the log-likelihood becomes:
\begin{align*}
\Likelihood(\Vparam|\VY,\VX) =& \textcolor{\darkred}{ \frac{p}{2} \log(2\pi)-\frac{1}{2} \log\left(\left|\sum_{i=1}^n w^*_i \VX_i \Omega_i^{-1}(\Vparam) \trans{\VX}_i\right|\right)} \notag \\
& + \sum_{i=1}^{n}  \left(\textcolor{\darkblue}{-\frac{m}{2}w_i\log\left(\frac{2\pi}{r_i}\right)- \frac{w_i}{2} \log\left|\Omega_i(\Vparam)\right| - \frac{w_i^*}{2} \Vvarepsilon_i(\Vparam) \Omega_i(\Vparam)^{-1} \trans{\Vvarepsilon_i(\Vparam)}} \right) 
\end{align*}
The expression of the score, Hessian, degrees of freedom can be
modified similarly by using the modified \(w_i^*\) weight for the REML
and residual terms (but not for the ML trace terms).
